<?xml version="1.0" encoding="UTF-8"?>
<definitions xmlns="https://www.omg.org/spec/DMN/20191111/MODEL/"
             xmlns:dmndi="https://www.omg.org/spec/DMN/20191111/DMNDI/"
             xmlns:dc="http://www.omg.org/spec/DMN/20180521/DC/"
             xmlns:di="http://www.omg.org/spec/DMN/20180521/DI/"
             id="Definitions_TaskPriority"
             name="TaskPriorityModel"
             namespace="http://smartstudyplanner.example/dmn">

  <inputData id="InputData_urgency" name="urgency"/>
  <inputData id="InputData_effort" name="effort"/>
  <inputData id="InputData_deadline" name="hasDeadline"/>

  <decision id="Decision_TaskPriority" name="Определение приоритета задачи">
    <informationRequirement id="IR_urgency">
      <requiredInput href="#InputData_urgency"/>
    </informationRequirement>

    <informationRequirement id="IR_effort">
      <requiredInput href="#InputData_effort"/>
    </informationRequirement>

    <informationRequirement id="IR_deadline">
      <requiredInput href="#InputData_deadline"/>
    </informationRequirement>

    <decisionTable id="DecisionTable_TaskPriority" hitPolicy="FIRST">
      <input id="InputClause_urgency" label="Срочность">
        <inputExpression id="InputExpression_urgency" typeRef="string">
          <text>urgency</text>
        </inputExpression>
        <inputValues id="InputValues_urgency">
          <text>"high","medium","low"</text>
        </inputValues>
      </input>

      <input id="InputClause_effort" label="Трудоемкость">
        <inputExpression id="InputExpression_effort" typeRef="string">
          <text>effort</text>
        </inputExpression>
        <inputValues id="InputValues_effort">
          <text>"large","medium","small"</text>
        </inputValues>
      </input>

      <input id="InputClause_deadline" label="Есть фиксированный дедлайн">
        <inputExpression id="InputExpression_deadline" typeRef="boolean">
          <text>hasDeadline</text>
        </inputExpression>
      </input>

      <output id="OutputClause_priority"
              name="priority"
              label="Приоритет"
              typeRef="string">
        <outputValues id="OutputValues_priority">
          <text>"high","medium","low"</text>
        </outputValues>
      </output>

      <annotation name="Пояснение"/>

      <rule id="Rule_1">
        <inputEntry id="Rule_1_Input_1"><text>"high"</text></inputEntry>
        <inputEntry id="Rule_1_Input_2"><text>-</text></inputEntry>
        <inputEntry id="Rule_1_Input_3"><text>-</text></inputEntry>
        <outputEntry id="Rule_1_Output_1"><text>"high"</text></outputEntry>
        <annotationEntry id="Rule_1_Annotation"><text>Высокая срочность всегда дает высокий приоритет.</text></annotationEntry>
      </rule>

      <rule id="Rule_2">
        <inputEntry id="Rule_2_Input_1"><text>"medium"</text></inputEntry>
        <inputEntry id="Rule_2_Input_2"><text>"large"</text></inputEntry>
        <inputEntry id="Rule_2_Input_3"><text>true</text></inputEntry>
        <outputEntry id="Rule_2_Output_1"><text>"high"</text></outputEntry>
        <annotationEntry id="Rule_2_Annotation"><text>Средняя срочность при большой трудоемкости и фиксированном дедлайне повышается до высокого приоритета.</text></annotationEntry>
      </rule>

      <rule id="Rule_3">
        <inputEntry id="Rule_3_Input_1"><text>"medium"</text></inputEntry>
        <inputEntry id="Rule_3_Input_2"><text>-</text></inputEntry>
        <inputEntry id="Rule_3_Input_3"><text>true</text></inputEntry>
        <outputEntry id="Rule_3_Output_1"><text>"medium"</text></outputEntry>
        <annotationEntry id="Rule_3_Annotation"><text>Средняя срочность с фиксированным дедлайном сохраняет средний приоритет.</text></annotationEntry>
      </rule>

      <rule id="Rule_4">
        <inputEntry id="Rule_4_Input_1"><text>"medium"</text></inputEntry>
        <inputEntry id="Rule_4_Input_2"><text>"small"</text></inputEntry>
        <inputEntry id="Rule_4_Input_3"><text>false</text></inputEntry>
        <outputEntry id="Rule_4_Output_1"><text>"medium"</text></outputEntry>
        <annotationEntry id="Rule_4_Annotation"><text>Средняя срочность без фиксированного дедлайна и с малой трудоемкостью остается средней.</text></annotationEntry>
      </rule>

      <rule id="Rule_5">
        <inputEntry id="Rule_5_Input_1"><text>"low"</text></inputEntry>
        <inputEntry id="Rule_5_Input_2"><text>-</text></inputEntry>
        <inputEntry id="Rule_5_Input_3"><text>true</text></inputEntry>
        <outputEntry id="Rule_5_Output_1"><text>"medium"</text></outputEntry>
        <annotationEntry id="Rule_5_Annotation"><text>Низкая срочность при наличии фиксированного дедлайна повышается до среднего приоритета.</text></annotationEntry>
      </rule>

      <rule id="Rule_6">
        <inputEntry id="Rule_6_Input_1"><text>"low"</text></inputEntry>
        <inputEntry id="Rule_6_Input_2"><text>-</text></inputEntry>
        <inputEntry id="Rule_6_Input_3"><text>false</text></inputEntry>
        <outputEntry id="Rule_6_Output_1"><text>"low"</text></outputEntry>
        <annotationEntry id="Rule_6_Annotation"><text>Низкая срочность без дедлайна сохраняет низкий приоритет.</text></annotationEntry>
      </rule>

      <rule id="Rule_7">
        <inputEntry id="Rule_7_Input_1"><text>-</text></inputEntry>
        <inputEntry id="Rule_7_Input_2"><text>-</text></inputEntry>
        <inputEntry id="Rule_7_Input_3"><text>-</text></inputEntry>
        <outputEntry id="Rule_7_Output_1"><text>"low"</text></outputEntry>
        <annotationEntry id="Rule_7_Annotation"><text>Правило по умолчанию: если параметры не заданы или не соответствуют другим правилам, устанавливается низкий приоритет.</text></annotationEntry>
      </rule>
    </decisionTable>
  </decision>

  <dmndi:DMNDI>
    <dmndi:DMNDiagram id="DMNDiagram_1" name="Task Priority DRD">
      <dmndi:DMNShape id="DMNShape_urgency" dmnElementRef="InputData_urgency">
        <dc:Bounds x="100" y="120" width="180" height="50"/>
      </dmndi:DMNShape>

      <dmndi:DMNShape id="DMNShape_effort" dmnElementRef="InputData_effort">

      <dmndi:DMNShape id="DMNShape_deadline" dmnElementRef="InputData_deadline">

      <dmndi:DMNShape id="DMNShape_decision" dmnElementRef="Decision_TaskPriority">
        <dc:Bounds x="420" y="220" width="260" height="100"/>
      </dmndi:DMNShape>

      <dmndi:DMNEdge id="DMNEdge_urgency" dmnElementRef="IR_urgency">
        <di:waypoint x="280" y="145"/>
        <di:waypoint x="420" y="250"/>
      </dmndi:DMNEdge>

      <dmndi:DMNEdge id="DMNEdge_effort" dmnElementRef="IR_effort">

        <di:waypoint x="420" y="270"/>

        <di:waypoint x="420" y="290"/>
      </dmndi:DMNEdge>
    </dmndi:DMNDiagram>
  </dmndi:DMNDI>

</definitions>
